const executorInput = document.getElementById('executor-input'); const chatSendButton = document.getElementById('chat-send'); const executorSendButton = document.getElementById('executor-send'); const stepButton = document.getElementById('step-button'); const resetButton = document.getElementById('reset-button'); const phaseDisplay = document.getElementById('current-phase'); // Add message to chat function addMessage(container, message, sender, includeCode = false) { const messageDiv = document.createElement('div'); messageDiv.className = `message ${sender}`; messageDiv.textContent = message; const timestamp = document.createElement('div'); timestamp.className = 'timestamp'; timestamp.textContent = new Date().toLocaleTimeString(); messageDiv.appendChild(timestamp); if (includeCode && workflow[currentStep].code) { const codeDiv = document.createElement('div'); codeDiv.className = 'code'; codeDiv.textContent = workflow[currentStep].code; messageDiv.appendChild(codeDiv); } container.appendChild(messageDiv); container.scrollTop = container.scrollHeight; } // Process next step in workflow function processNextStep() { if (currentStep >= workflow.length) { alert('Workflow complete!'); return; } const step = workflow[currentStep]; phaseDisplay.textContent = step.phase; switch (step.action) { case 'chat-to-executor': addMessage(chatMessages, step.message, step.sender); addMessage(executorMessages, `From frame1: ${step.message}`, 'user'); break; case 'executor-to-chat': addMessage(executorMessages, step.message, step.sender, true); addMessage(chatMessages, `From frame2: ${step.message}`, 'user'); break; case 'chat-to-user': addMessage(chatMessages, step.message, step.sender); break; } currentStep++; } // Event listeners stepButton.addEventListener('click', processNextStep); resetButton.addEventListener('click', () => { // Reset the chat messages chatMessages.innerHTML = '';
